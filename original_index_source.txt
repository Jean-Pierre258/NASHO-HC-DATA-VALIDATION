<script>
let token = '';
let currentUser = null;
const AUTH_STORAGE_KEY = 'NASHO_HC_AUTH_SESSION_V1';
const IDLE_LIMIT_MS = 10 * 60 * 1000;
const HEARTBEAT_INTERVAL_MS = 2 * 60 * 1000;
let idleTimer = null;
let lastHeartbeatAt = 0;
let restoringSession = false;

const $ = id => document.getElementById(id);

/* LOGIN TABS */

function switchLogin(type) {
  const staff = type === 'staff';

  $('staffLoginForm').classList.toggle('hidden', !staff);
  $('adminLoginForm').classList.toggle('hidden', staff);

  $('staffTab').classList.toggle('active', staff);
  $('adminTab').classList.toggle('active', !staff);

  $('message').classList.add('hidden');
}

function showMessage(text, error=false) {
  $('message').textContent = text;
  $('message').className = 'status' + (error ? ' error' : '');
  $('message').classList.remove('hidden');
}

function handleError(error) {
  const message = error && error.message
    ? error.message
    : String(error || 'An unexpected error occurred.');
  showMessage(message, true);
}

/* GOOGLE APPS SCRIPT CONNECTION */

const APPSCRIPT_URL =
  'https://script.google.com/macros/s/AKfycbxPKKE6bdYC6CEGio04KjmRQ6G7TPjv272EpsFiJDYlVmpiv45U6cW4lRYEKLnLk0b3/exec';

/* API CONNECTION */

async function api(functionName, args = [], callback) {
  try {
    const response = await fetch(APPSCRIPT_URL, {
      method: 'POST',
      redirect: 'follow',
      headers: {
        'Content-Type': 'text/plain;charset=utf-8'
      },
      body: JSON.stringify({
        action: functionName,
        args: args
      })
    });

    let result;

    try {
      result = await response.json();
    } catch (_) {
      throw new Error(
        'Google Apps Script returned an invalid response. Check your deployment and doPost(e).'
      );
    }

    if (!response.ok || !result || result.success !== true) {
      const errorMessage = (result && result.error) ||
        ('Request failed with HTTP ' + response.status);
      if (/session (expired|token is missing)/i.test(errorMessage) && token) {
        forceLogout('Your session expired. Please log in again.');
      }
      throw new Error(errorMessage);
    }

    if (typeof callback === 'function') {
      callback(result.data);
    }

    return result.data;

  } catch (error) {
    handleError(error);
    return null;
  }
}

/* LOGIN */

function requestCode() {
  const identifier = $('identifier').value.trim();

  if (!identifier) {
    showMessage('Enter your registered email or phone number.', true);
    return;
  }

  api('requestLoginCode', [identifier], showMessage);
}

function verifyCode() {
  const identifier = $('identifier').value.trim();
  const otp = $('otp').value.trim();

  if (!identifier || !otp) {
    showMessage('Enter your registered details and verification code.', true);
    return;
  }

  api(
    'verifyLoginCode',
    [identifier, otp],
    enterApplication
  );
}

function adminLogin() {
  const username = $('adminUsername').value.trim();
  const password = $('adminPassword').value;

  if (!username || !password) {
    showMessage('Enter administrator username and password.', true);
    return;
  }

  const button = $('adminLoginForm').querySelector('button.login-submit');
  if (button) {
    button.disabled = true;
    button.textContent = 'Signing in...';
  }

  api(
    'adminLogin',
    [username, password],
    session => {
      if (button) {
        button.disabled = false;
        button.textContent = 'Login as Administrator';
      }
      enterApplication(session);
    }
  );

  window.setTimeout(() => {
    if (button && document.body.classList.contains('login-active')) {
      button.disabled = false;
      button.textContent = 'Login as Administrator';
    }
  }, 15000);
}

function enterApplication(session) {
  if (!session || !session.token || !session.user) {
    showMessage('Invalid login response. Contact the administrator.', true);
    return;
  }

  token = session.token;
  currentUser = session.user;
  try {
    localStorage.setItem(AUTH_STORAGE_KEY, JSON.stringify({
      token: token,
      user: currentUser
    }));
  } catch (storageError) {
    console.warn('Could not persist the login session in this browser.', storageError);
  }
  startIdleProtection();
  lastHeartbeatAt = Date.now();

  document.body.classList.remove('login-active');

  $('login').classList.add('hidden');
  $('application').classList.remove('hidden');
  $('profile').classList.remove('hidden');

  $('who').textContent = currentUser.fullName;
  $('role').textContent =
    currentUser.role + ' | ' + (currentUser.service || 'All Services');

  $('avatar').src = currentUser.photoURL ||
    'https://www.gravatar.com/avatar/?d=mp';

  $('adminPanel').classList.toggle(
    'hidden',
    currentUser.role !== 'Admin'
  );

  $('representativeAssignment').classList.toggle(
    'hidden',
    currentUser.role !== 'Admin'
  );

  refreshHistory();

  if (currentUser.role === 'Admin') {
    loadAdmin();
  }
}

function clearStoredSession() {
  try { localStorage.removeItem(AUTH_STORAGE_KEY); } catch (_) {}
}

function stopIdleProtection() {
  if (idleTimer) window.clearTimeout(idleTimer);
  idleTimer = null;
}

function forceLogout(message) {
  const oldToken = token;
  token = '';
  currentUser = null;
  stopIdleProtection();
  clearStoredSession();

  if (oldToken) {
    // Best-effort server-side invalidation; never block the local lock screen.
    api('logout', [oldToken]);
  }

  $('application').classList.add('hidden');
  $('profile').classList.add('hidden');
  $('login').classList.remove('hidden');
  document.body.classList.add('login-active');

  if (message) showMessage(message, true);
}

function resetIdleTimer() {
  if (!token || !currentUser) return;
  if (idleTimer) window.clearTimeout(idleTimer);
  idleTimer = window.setTimeout(() => {
    forceLogout('You were signed out after 10 minutes of inactivity. Please log in again.');
  }, IDLE_LIMIT_MS);

  // Refresh the server-side inactivity timestamp periodically while the user
  // is actively using the page, not on every mouse movement.
  const now = Date.now();
  if (now - lastHeartbeatAt >= HEARTBEAT_INTERVAL_MS) {
    lastHeartbeatAt = now;
    api('touchSession', [token], result => {
      if (!result || result.valid !== true) {
        forceLogout('Your session expired. Please log in again.');
      }
    });
  }
}

function startIdleProtection() {
  stopIdleProtection();
  resetIdleTimer();
}

function restoreSavedSession() {
  if (restoringSession) return;
  restoringSession = true;

  let saved = null;
  try {
    saved = JSON.parse(localStorage.getItem(AUTH_STORAGE_KEY) || 'null');
  } catch (_) {
    clearStoredSession();
  }

  if (!saved || !saved.token) {
    restoringSession = false;
    return;
  }

  (async () => {
    const result = await api('validateSession', [saved.token]);
    if (result && result.valid && result.user) {
      enterApplication({
        token: saved.token,
        user: result.user
      });
    } else {
      clearStoredSession();
      showMessage('Your session expired. Please log in again.', true);
    }
    restoringSession = false;
  })();
}

function logout() {
  const oldToken = token;
  token = '';
  currentUser = null;
  stopIdleProtection();
  clearStoredSession();
  if (oldToken) {
    api('logout', [oldToken], () => {
      $('application').classList.add('hidden');
      $('profile').classList.add('hidden');
      $('login').classList.remove('hidden');
      document.body.classList.add('login-active');
      showMessage('You have signed out successfully.');
    });
  } else {
    forceLogout('You have signed out successfully.');
  }
}

// Any real user activity restarts the ten-minute inactivity countdown.
['click', 'keydown', 'mousemove', 'scroll', 'touchstart', 'input'].forEach(eventName => {
  document.addEventListener(eventName, () => {
    if (token) resetIdleTimer();
  }, { passive: true });
});

// Restore a valid session after refresh instead of returning to the login page.
document.addEventListener('DOMContentLoaded', restoreSavedSession);

/* HISTORY */

function refreshHistory() {
  api('listEvaluations', [token], evaluations => {

    $('historyTable').innerHTML = evaluations.map(e => `
      <tr>
        <td>${e.evaluationDate}</td>
        <td>${escapeValue(e.service)}</td>
        <td>${escapeValue(e.representativeName)}</td>
        <td>${escapeValue(e.status)}</td>
        <td>
          ${e.status === 'Submitted' || e.status === 'Completed'
            ? `<button class="secondary" onclick="viewReport('${e.evaluationId}')">View Report</button>

function localTodayISO() {
  const d = new Date();
  const offset = d.getTimezoneOffset();
  return new Date(d.getTime() - offset * 60000).toISOString().slice(0,10);
}

function fileToDataUrl(file) {
  return new Promise((resolve, reject) => {
    if (!file) return resolve('');
    if (!file.type || !file.type.startsWith('image/')) {
      reject(new Error('Signature must be an image file.'));
      return;
    }
    if (file.size > 2 * 1024 * 1024) {
      reject(new Error('Signature image must be 2 MB or smaller.'));
      return;
    }
    const reader = new FileReader();
    reader.onload = () => resolve(String(reader.result || ''));
    reader.onerror = () => reject(new Error('Could not read signature image.'));
    reader.readAsDataURL(file);
  });
}

function escapeValue(value) {
  return String(value ?? '').replace(/[&<>"']/g, c => ({
    '&':'&amp;',
    '<':'&lt;',
    '>':'&gt;',
    '"':'&quot;',
    "'":'&#39;'
  })[c]);
}

function recalculateDataRow(row) {
  const rText = row.querySelector('.reportedValue').value.trim();
  const vText = row.querySelector('.verifiedValue').value.trim();
  const r = Number(rText), v = Number(vText);
  row.querySelector('.discrepancy').value =
    rText !== '' && vText !== '' && Number.isFinite(r) && Number.isFinite(v) && r !== 0
      ? (((v - r) / r) * 100).toFixed(2) + '%' : '';
  row.querySelector('.gaps').value =
    rText !== '' && vText !== '' && Number.isFinite(r) && Number.isFinite(v)
      ? String(v - r) : '';
}

function collectRows(container, mapping) {
  return [...document.querySelectorAll('#' + container + ' .item')]
    .map(row => {
      const result = {};

      Object.keys(mapping).forEach(key => {
        result[key] =
          row.querySelector('.' + mapping[key]).value;
      });

      return result;
    });
}

function reportRows(rows, fields) {
  if (!rows || !rows.length) return '<tr><td colspan="' + fields.length + '">&nbsp;</td></tr>'.repeat(5);
  return rows.map(row => '<tr>' + fields.map(f =>
    '<td>' + escapeValue(row[f] ?? '') + '</td>'
  ).join('') + '</tr>').join('');
}

function viewReport(id) {
  const win = window.open('', '_blank');
  if (!win) { showMessage('Please allow pop-ups to view the report.', true); return; }
  api('getEvaluation', [token, id], result => {
    win.document.open();
    win.document.write(buildReportHtml(result));

  });
}

function openEvaluation(id) {
  api('getEvaluation', [token, id], result => {
    const e = result.evaluation;
    if (e.status === 'Submitted' || e.status === 'Completed') {
      const win = window.open('', '_blank');
      if (!win) { showMessage('Please allow pop-ups to view the report.', true); return; }
      win.document.open(); win.document.write(buildReportHtml(result)); win.document.close();
      return;
    }
    $('evaluationId').value = id;
    $('service').value = e.service || '';
    $('healthCenter').value = e.healthCenter || 'NASHO HEALTH CENTER';
    $('evaluationDate').value = String(e.evaluationDate || '').slice(0,10);
    $('responsibleName').value = e.representativeName || currentUser.fullName || '';
    $('responsibleSignature').value = '';
    const mainPreview = $('responsibleSignaturePreview');
    mainPreview.src = e.responsibleSignatureDataUrl || '';
    mainPreview.classList.toggle('hidden', !e.responsibleSignatureDataUrl);
    $('verifiedRows').innerHTML = '';
    $('dataRows').innerHTML = '';
    $('toolRows').innerHTML = '';
    (e.verifiedBy || []).forEach(addVerifiedRow);
    if (!(e.verifiedBy || []).length) addVerifiedRow();
    result.items.forEach(addDataRow);
    result.tools.forEach(addToolRow);
    $('evaluationForm').classList.remove('hidden');
    setFormReadOnly(false);
  });
}
